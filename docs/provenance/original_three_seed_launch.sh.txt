COMPLETION="$ROOT/12_three_seed_completion_272_v1"

STATUS="$QUEUE/status.tsv"
CONFIG="$ASSETS/20260803_aug_jitter005_mask010_seed42.yaml"
PRETRAIN=experiments/experiments/20260629_fcmae_convnext_eca_pretrain/fcmae_convnext_eca_best.pt
PRETRAIN_NO_ECA=experiments/experiments/20260623_fcmae_convnext/fcmae_convnext_best.pt

record() {
  printf '%s\t%s\t%s\n' "$(date -Iseconds)" "$1" "$2" >>"$STATUS"
  printf '[%s] %s %s\n' "$(date -Iseconds)" "$1" "$2"
}
on_exit() {
  code=$?
  trap - EXIT INT TERM
  if [ "$code" -ne 0 ]; then
    record FAILED "exit_code=$code shutdown=required"
    printf '%s\n' "$code" >"$QUEUE/QUEUE_FAILED"
    sync
    printf '%s\n' "$(date -Iseconds)" >"$QUEUE/SHUTDOWN_AFTER_FAILURE_REQUESTED"
    /usr/bin/shutdown -h now || true
  fi
  exit "$code"
}
trap on_exit EXIT INT TERM

export PYTHONDONTWRITEBYTECODE=1
export PYTHONPATH="$SNAPSHOT:$SNAPSHOT/src${PYTHONPATH:+:$PYTHONPATH}"
export PYTHONUNBUFFERED=1 CUDA_VISIBLE_DEVICES=0

test "$(git rev-parse 'HEAD^{tree}')" = "$EXPECTED_TREE"
test -z "$(git status --porcelain)"

fi

  printf 'launch=%s\nserver=272\nseeds=42,2026,3407\naugmentation=jitter0.05,mask0.1\n' "$(date -Iseconds)"

  printf 'reduced_leads=limb6,precordial6,anatomical4,lead_i,lead_ii; 12lead_pretrain+slice+stem_select; no V1\n'
  printf 'analyses=predictions,per-label,paired bootstrap 2000,95CI,subgroups,efficiency,cross-seed attribution,shuffled-label sanity\n'
  sha256sum "$CONFIG" "$PRETRAIN" "$PRETRAIN_NO_ECA" "$ASSETS"/*.py "$ASSETS/remote_queue_aug_three_seed_remaining_272_20260820.sh"
} >"$COMPLETION/source_manifest.txt"

"$PY" - "$COMPLETION/existing_results_not_retrained.json" <<'PY'

payload = {
    "policy": "existing completed results are not retrained",
    "not_retrained": {
        "full": {"seeds": [42, 2026, 3407], "reason": "metrics and exactly one resolved checkpoint per seed already exist"},
        "low_label": {
            "experiments": ["ssl_01pct", "scratch_01pct", "ssl_05pct", "scratch_05pct", "ssl_10pct", "scratch_10pct", "ssl_30pct", "scratch_30pct"],
            "seeds": [42, 2026, 3407],
            "reason": "completed previously on server 15866; user explicitly prohibited continuing existing results",

        "seed42_core_and_components": {"seeds": [42], "reason": "validated existing rows and checkpoints"},
        "efficiency": {"seed": 42, "reason": "model efficiency is deterministic for the fixed architecture and already complete"},
    },
    "low_label_100pct_aliases": {"ssl": "01_core/full", "scratch": "01_core/strict_no_pretraining"},
}
open(sys.argv[1], "w", encoding="utf-8").write(json.dumps(payload, indent=2))
PY

metric_seeds() {
  "$PY" - "$1/metrics.csv" <<'PY'
import pandas as pd,sys
try:
    frame=pd.read_csv(sys.argv[1])
    print(" ".join(str(value) for value in sorted(set(frame.Seed.astype(int)))))
except FileNotFoundError:
    print("")
PY
}
missing_seeds() {
  "$PY" - "$1/metrics.csv" <<'PY'
import pandas as pd,sys
expected=[42,2026,3407]
try: present=set(pd.read_csv(sys.argv[1]).Seed.astype(int))
except FileNotFoundError: present=set()
print(" ".join(str(seed) for seed in expected if seed not in present))
PY
}
validate_metrics() {
  "$PY" - "$1/metrics.csv" <<'PY'
import pandas as pd,sys
frame=pd.read_csv(sys.argv[1])
assert len(frame)==3 and set(frame.Seed.astype(int))=={42,2026,3407}, (len(frame),frame.Seed.tolist())
PY
}
checkpoint_for_seed() {
  matches=("$1"/*seed"$2".pth)
  test "${#matches[@]}" -eq 1
  test -e "${matches[0]}"
  printf '%s\n' "${matches[0]}"
}
run_downstream_missing() {
  target=$1 experiment_name=$2 description=$3
  shift 3
  read -r -a missing <<<"$(missing_seeds "$target")"
  if [ "${#missing[@]}" -eq 0 ]; then
    record SKIP "$experiment_name training seeds=$(metric_seeds "$target")"
    validate_metrics "$target"
    return
  fi
  record START "$experiment_name missing_seeds=${missing[*]}"
  "$PY" scripts/train_downstream.py --config "$CONFIG" --seeds "${missing[@]}" --num-workers 8 \
    --output-root "$(dirname "$target")" --experiment-name "$experiment_name" --des "$description" "$@"
  validate_metrics "$target"
  record COMPLETE "$experiment_name seeds=42,2026,3407"
}
run_tabular_missing() {
  target=$1
  read -r -a missing <<<"$(missing_seeds "$target")"
  if [ "${#missing[@]}" -eq 0 ]; then record SKIP "tabular_only training"; validate_metrics "$target"; return; fi
  record START "tabular_only missing_seeds=${missing[*]}"
  "$PY" scripts/train_tabular_only.py --dataset echonext --data-dir data/echonext --tabular-features 7 \
    --epochs 100 --patience 10 --batch-size 256 --num-workers 8 --seeds "${missing[@]}" \
    --base-lr 0.001 --weight-decay 0.0001 --output-root "$(dirname "$target")" \
    --experiment-name tabular_only --des tabular_only_waveform_augmentation_not_applicable
  validate_metrics "$target"
  record COMPLETE "tabular_only seeds=42,2026,3407"
}
export_predictions() {
  target=$1 lead=${2:-full12} mode=${3:-zero_fill} exporter=${4:-wave}
  mkdir -p "$target/predictions"
  for seed in "${SEEDS[@]}"; do
    checkpoint=$(checkpoint_for_seed "$target" "$seed")
    for split in val test; do
      output="$target/predictions/seed_${seed}_${split}.npz"
      if [ -s "$output" ] && [ -s "${output%.npz}.manifest.json" ]; then continue; fi
      record START "export $(basename "$target") seed=$seed split=$split"
      if [ "$exporter" = tabular ]; then
        "$PY" "$ASSETS/export_tabular_predictions.py" --checkpoint "$checkpoint" --seed "$seed" --split "$split" \
          --data-dir data/echonext --wave-name "EchoNext_${split}_waveforms.npy" \
          --tab-name "EchoNext_${split}_tabular_features.npy" --output "$output" --batch-size 512 --num-workers 8
      else
        "$PY" scripts/export_shd_predictions.py --checkpoint "$checkpoint" --seed "$seed" --split "$split" \
          --data-dir data/echonext --wave-name "EchoNext_${split}_waveforms.npy" \
          --tab-name "EchoNext_${split}_tabular_features.npy" --output "$output" --batch-size 512 --num-workers 8 \
          --lead-setting "$lead" --lead-input-mode "$mode"
      fi
    done
  done
}
review_valid() {
  "$PY" - "$1/review_status.json" "$2" <<'PY'
import json,sys
status=json.load(open(sys.argv[1],encoding="utf-8"))
assert status.get("status")=="complete"
assert set(status.get("seeds",[]))=={42,2026,3407}
if sys.argv[2]=="paired": assert status.get("bootstrap_iterations")==2000
PY
}
review_experiment() {
  target=$1 reference=${2:-}
  mode=unpaired; [ -n "$reference" ] && mode=paired
  if [ -s "$target/review_status.json" ] && review_valid "$target" "$mode" 2>/dev/null; then
    record SKIP "review $(basename "$target")"
    return
  fi
  cmd=("$PY" analysis/review_experiments_v1.py --experiment-dir "$target" --expected-seeds 42 2026 3407 --bootstrap-iters 2000 --bootstrap-seed 20260820)
  [ -n "$reference" ] && cmd+=(--reference-template "$reference")
  record START "review $(basename "$target")"
  "${cmd[@]}"
  review_valid "$target" "$mode"
}
